import pandas as pd
import joblib

from pathlib import Path

from sklearn.model_selection import train_test_split
from sklearn.pipeline import Pipeline
from sklearn.impute import SimpleImputer
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier

from sklearn.metrics import (
    classification_report,
    confusion_matrix,
    roc_auc_score,
    average_precision_score
)

# 1. Load dataset
df = pd.read_csv("training_data.csv")

# 2. Define input features
FEATURES = [
    "session_duration",
    "mouse_distance",
    "mouse_speed_mean",
    "mouse_speed_std",
    "click_count",
    "avg_click_interval",
    "request_count",
    "request_rate",
    "interaction_count",
    "interaction_interval_std"
]

TARGET = "label"

# 3. Separate features and target
X = df[FEATURES]
y = df[TARGET]

# Validate labels
if set(y.unique()) != {0, 1}:
    raise ValueError("Dataset must contain both classes: 0 and 1")

# 4. Split training and testing data
X_train, X_test, y_train, y_test = train_test_split(
    X,
    y,
    test_size=0.20,
    random_state=42,
    stratify=y
)

# 5. Create Logistic Regression pipeline
lr_model = Pipeline([
    ("imputer", SimpleImputer(strategy="median")),
    ("scaler", StandardScaler()),
    ("classifier", LogisticRegression(
        max_iter=1000,
        class_weight="balanced"
    ))
])

# 6. Create Random Forest pipeline
rf_model = Pipeline([
    ("imputer", SimpleImputer(strategy="median")),
    ("classifier", RandomForestClassifier(
        n_estimators=200,
        max_depth=12,
        random_state=42,
        class_weight="balanced"
    ))
])

# 7. Train both models
models = {
    "Logistic Regression": lr_model,
    "Random Forest": rf_model
}

Path("exported_models").mkdir(exist_ok=True)

for name, model in models.items():

    print(f"\nTraining {name}...")

    model.fit(X_train, y_train)

    # 8. Predict
    predictions = model.predict(X_test)

    probabilities = model.predict_proba(X_test)[:, 1]

    # 9. Evaluate
    print("\nClassification Report:")
    print(classification_report(
        y_test,
        predictions,
        zero_division=0
    ))

    print("Confusion Matrix:")
    print(confusion_matrix(y_test, predictions))

    print("ROC-AUC:", roc_auc_score(
        y_test, probabilities
    ))

    print("PR-AUC:", average_precision_score(
        y_test, probabilities
    ))

    # 10. Export model
    filename = (
        "logistic_regression.joblib"
        if name == "Logistic Regression"
        else "random_forest.joblib"
    )

    joblib.dump(
        {
            "pipeline": model,
            "features": FEATURES,
            "positive_class": 1
        },
        f"exported_models/{filename}"
    )

    print(f"{name} saved successfully!")



Step	What happens
1	Reads our CSV dataset
2	Selects the behavioral features
3	Separates inputs from labels
4	Divides data into training and testing sets
5	Creates Logistic Regression
6	Creates Random Forest
7	Trains both models
8	Generates predictions
9	Evaluates model performance
10	Saves trained models
